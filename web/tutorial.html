float e = d_left - d_right;                 // our choice: left minus right
// keep the sign consistent with the wheels:  v_left = base - u, v_right = base + u
integral += e * dt;                         // I: the past
float d = (e - prev_e) / dt;                // D: the future (slope)
prev_e = e;
float u = kp * e + ki * integral + kd * d;  // P: the present

integral = clamp(integral + e * dt, -i_limit, i_limit);       // anti-windup
d_filt   = alpha * (e - prev_e) / dt + (1 - alpha) * d_filt;  // low-pass on D
